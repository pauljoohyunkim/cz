<program>            ::= <global_declaration_list>

<global_declaration_list> ::= <global_declaration> <global_declaration_list> | ""

<global_declaration> ::= <function_decl>
                       | <struct_decl>
                       | <variable_decl> ";"
                       | <typedef_decl> ";"
                       | <newtype_decl> ";"

;; --- Type Aliasing ---
<typedef_decl>       ::= "typedef" <type> <identifier>
<newtype_decl>       ::= "newtype" <type> <identifier>
                       
;; --- Structural Statements & Blocks ---
<block>            ::= "{" <statement_list> "}"
<statement_list>   ::= <statement> <statement_list> | ""

<statement>        ::= <variable_decl> ";"
                     | <assignment_stmt> ";"
                     | <return_stmt> ";"
                     | <if_stmt>
                     | <for_stmt>
                     | <while_stmt>
                     | <block>
                     | <expression> ";"

;; --- Conditionals (if / else if / else) ---
<if_stmt>           ::= "if" "(" <expression> ")" <block> [ "else" <else_branch> ]
<else_branch>       ::= <if_stmt> | <block>

;; --- Loops (while / for) ---
<while_stmt>        ::= "while" "(" <expression> ")" <block>
<for_stmt>          ::= "for" "(" [ <variable_decl> | <assignment_stmt> ] ";" [ <expression> ] ";" [ <assignment_stmt> | <expression> ] ")" <block>

;; --- Declarations & Assignments ---
<variable_decl>    ::= <param_decl> [ "=" <expression> ]

<assignment_stmt>  ::= <expression> <assignment_op> <expression>
<assignment_op>    ::= "=" | "+=" | "-=" | "*=" | "/=" | "%=" | "&=" | "|=" | "^="

<return_stmt>      ::= "return" [ <expression> ]

;; --- Top-Level Structural Definitions ---
<function_decl>    ::= "func" <identifier> "::" "(" <param_list> ")" [ "->" <type> ] <block>
<param_list>       ::= <param_decl> | <param_decl> "," <param_list> | ""
<param_decl>       ::= <identifier> "::" <type>

<struct_decl>      ::= "struct" <identifier> "{" <struct_member_list> "}"
<struct_member_list> ::= <struct_member> ";" <struct_member_list> | ""
<struct_member>    ::= <variable_decl>

<struct_init>      ::= <identifier> "{" <struct_init_list> "}"
<struct_init_list>   ::= [ <struct_init_member> ( "," <struct_init_member> )* [ "," ] ]
<struct_init_member> ::= "." <identifier> "=" <expression>

<array_init>       ::= "[" <elem_list> "]"
<elem_list>        ::= <primary> | <primary> "," <elem_list> | ""

;; --- Types (Supporting Explicit References) ---
<mutability>       ::= "const" | ""
<type>             ::= <mutability> <base_type> [ <array_specifier> ] [ "&" ]
<base_type>        ::= "int32" | "uint32" | "bool" | "float" | <identifier> | | "(" <type> ")"
<array_specifier>  ::= "[" [ <expression> ] "]"


;; --- Expressions & Precedence Hierarchy (Fixed Association) ---
<expression>       ::= <equality>

<equality>         ::= <equality> "==" <relational> 
                     | <equality> "!=" <relational> 
                     | <relational>

<relational>       ::= <relational> "<"  <bitwise_or> 
                     | <relational> ">"  <bitwise_or> 
                     | <relational> "<=" <bitwise_or> 
                     | <relational> ">=" <bitwise_or> 
                     | <bitwise_or>

<bitwise_or>       ::= <bitwise_or> "|" <bitwise_xor> 
                     | <bitwise_xor>

<bitwise_xor>      ::= <bitwise_xor> "^" <bitwise_and> 
                     | <bitwise_and>

<bitwise_and>      ::= <bitwise_and> "&" <additive> 
                     | <additive>

<additive>         ::= <additive> "+" <multiplicative> 
                     | <additive> "-" <multiplicative> 
                     | <multiplicative>

<multiplicative>   ::= <multiplicative> "*" <cast_expr> 
                     | <multiplicative> "/" <cast_expr> 
                     | <multiplicative> "%" <cast_expr> 
                     | <cast_expr>

;; --- Type Casting (Right-Associative / High Precedence) ---
;; Binds tighter than multiplication, but looser than unary operations
<cast_expr>        ::= <unary> "as" <type> 
                     | <unary>

;; Unary operators naturally associate right-to-left (e.g., -!-x)
<unary>            ::= "-" <postfix> 
                     | "!" <postfix> 
                     | <postfix>

;; --- Postfix Operators (Left-Associative / Highest Operator Precedence) ---
<postfix>          ::= <postfix> "(" <argument_list> ")"   ;; Function Call
                     | <postfix> "." <identifier>          ;; Struct Member Access
                     | <postfix> "[" <expression> "]"      ;; Array/List Access
                     | <primary>

;; --- Primary Leaf Values ---
<primary>          ::= <identifier> 
                     | "NUMERICAL_LITERAL" 
                     | "STRING_LITERAL" 
                     | "TRUE"
                     | "FALSE"
                     | "(" <expression> ")" 
                     | <struct_init>

;; --- Argument List Helper ---
<argument_list>    ::= <expression> <argument_rest> | ""
<argument_rest>    ::= "," <expression> <argument_rest> | ""
